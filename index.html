            { note: 264, duration: 0.3 }, { note: 264, duration: 0.3 }, { note: 297, duration: 0.6 },
            { note: 264, duration: 0.6 }, { note: 352, duration: 0.6 }, { note: 330, duration: 1.0 },
            { note: 264, duration: 0.3 }, { note: 264, duration: 0.3 }, { note: 297, duration: 0.6 },
            { note: 264, duration: 0.6 }, { note: 396, duration: 0.6 }, { note: 352, duration: 1.0 }
        ];

        function playNote(freq, duration, time) {
            if (!audioCtx) return;
            const osc = audioCtx.createOscillator();
            const gain = audioCtx.createGain();
            
            osc.type = 'triangle';
            osc.frequency.value = freq;
            
            gain.gain.setValueAtTime(0.15, time);
            gain.gain.exponentialRampToValueAtTime(0.001, time + duration);
            
            osc.connect(gain);
            gain.connect(audioCtx.destination);
            
            osc.start(time);
            osc.stop(time + duration);
        }

        function playMelody() {
            if (!isPlaying) return;
            let now = audioCtx.currentTime;
            let delay = 0;

            notes.forEach(item => {
                let tid = setTimeout(() => {
                    if (isPlaying) playNote(item.note, item.duration, audioCtx.currentTime);
                }, delay * 1000);
                timeoutIds.push(tid);
                delay += item.duration + 0.1;
            });

            // Loop melody
            let loopTid = setTimeout(() => {
                if (isPlaying) playMelody();
            }, delay * 1000);
            timeoutIds.push(loopTid);
        }

        function toggleAudio() {
            initAudio();
            const musicText = document.getElementById('musicText');
            const musicPing = document.getElementById('musicPing');
            const musicDot = document.getElementById('musicDot');

            if (!isPlaying) {
                if (audioCtx.state === 'suspended') {
                    audioCtx.resume();
                }
                isPlaying = true;
                musicText.innerText = 'Playing';
                musicPing.classList.remove('hidden');
                musicDot.classList.replace('bg-gray-400', 'bg-roseGold');
                playMelody();
            } else {
                isPlaying = false;
                timeoutIds.forEach(id => clearTimeout(id));
                timeoutIds = [];
                musicText.innerText = 'Play Music';
                musicPing.classList.add('hidden');
                musicDot.classList.replace('bg-roseGold', 'bg-gray-400');
            }
        }

        // Toggle Letter Collapse
        function toggleLetter() {
            const letterBody = document.getElementById('letterBody');
            if (letterBody.classList.contains('hidden')) {
                letterBody.classList.remove('hidden');
            } else {
                letterBody.classList.add('hidden');
            }
        }
